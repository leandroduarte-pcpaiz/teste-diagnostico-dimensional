            t[
                "comprimento_maximo_mm"
            ]
            >=
            comprimento_longitudinal * 0.70
        )
    ]

    geometria_confirmada = (
        len(trilhos_relevantes) >= 3
    )

    return {
        "status":
            (
                "CONFIRMADO"
                if geometria_confirmada
                else
                "PARCIAL"
            ),

        "motivo":
            (
                "TRES_REGIOES_LONGITUDINAIS_IDENTIFICADAS"
                if geometria_confirmada
                else
                "REGIOES_DEFINIDAS_PELAS_DOBRAS_MAS_TRILHOS_INSUFICIENTES"
            ),

        "eixo_longitudinal":
            trilhos_info[
                "eixo_longitudinal"
            ],

        "comprimento_longitudinal_mm":
            comprimento_longitudinal,

        "dimensao_transversal_mm":
            dimensao_transversal,

        "trilhos":
            trilhos,

        "trilhos_relevantes":
            trilhos_relevantes,

        "pares_espessura":
            pares_espessura,

        "regioes_externas_mm":
            [
                regiao_1,
                regiao_2,
                regiao_3,
            ],

        "abas_identificadas":
            abas,
    }


# =============================================================================
# DETECTAR LINHAS DE DOBRA
# =============================================================================

def detectar_linhas_dobra_geometricas(
    segmentos: List[Segmento],
    contorno: Contorno,
    espessura_mm: Optional[float],
    raio_mm: Optional[float],
) -> dict:

    eixo = eixo_longitudinal(
        contorno
    )

    xmin, ymin, xmax, ymax = (
        bounding_box_segmentos(
            contorno.segmentos
        )
    )

    if eixo == "X":

        comprimento_longitudinal_pdf = (
            xmax - xmin
        )

        limite_transversal_min = ymin
        limite_transversal_max = ymax

    else:

        comprimento_longitudinal_pdf = (
            ymax - ymin
        )

        limite_transversal_min = xmin
        limite_transversal_max = xmax

    if comprimento_longitudinal_pdf <= 0:

        return {
            "encontradas": False,
            "quantidade": 0,
            "eixo_longitudinal": eixo,
            "eixo_desenvolvimento": (
                "Y" if eixo == "X" else "X"
            ),
            "candidatos": [],
            "linhas_selecionadas": [],
        }

    escala = contorno.escala

    if escala <= 0:
        escala = 1.0

    comprimento_longitudinal_mm = (
        comprimento_longitudinal_pdf *
        escala
    )

    distancia_borda_pdf = (
        MIN_DISTANCIA_BORDA_DOBRA_MM /
        escala
    )

    candidatos_brutos = []

    for segmento in segmentos:

        comprimento_pdf = (
            segmento.comprimento
        )

        if comprimento_pdf <= 0:
            continue

        comprimento_mm = (
            comprimento_pdf *
            escala
        )

        fracao = (
            comprimento_pdf /
            comprimento_longitudinal_pdf
        )

        if (
            fracao <
            MIN_FRACAO_COMPRIMENTO_LINHA_DOBRA
        ):
            continue

        angulo = segmento.angulo

        if eixo == "X":

            diferenca_angular = angulo_diferenca(
                angulo,
                0.0,
            )

        else:

            diferenca_angular = angulo_diferenca(
                angulo,
                90.0,
            )

        if (
            diferenca_angular >
            TOLERANCIA_ANGULO_DOBRA
        ):
            continue

        coord_transversal = (
            coordenada_transversal(
                segmento,
                eixo,
            )
        )

        if (
            coord_transversal <
            limite_transversal_min
        ):
            continue

        if (
            coord_transversal >
            limite_transversal_max
        ):
            continue

        distancia_borda = min(
            abs(
                coord_transversal -
                limite_transversal_min
            ),
            abs(
                limite_transversal_max -
                coord_transversal
            ),
        )

        if (
            distancia_borda <
            distancia_borda_pdf
        ):
            continue

        xlong1, xlong2 = (
            coordenada_longitudinal(
                segmento,
                eixo,
            )
        )

        # A linha de dobra deve estar efetivamente dentro do comprimento
        # longitudinal do contorno principal. Isso evita que cotas,
        # linhas de centro ou outras geometrias do desenho sejam
        # confundidas com linhas de dobra.
        if eixo == "X":
            limite_longitudinal_min = xmin
            limite_longitudinal_max = xmax
        else:
            limite_longitudinal_min = ymin
            limite_longitudinal_max = ymax

        sobreposicao = max(
            0.0,
            min(xlong2, limite_longitudinal_max)
            -
            max(xlong1, limite_longitudinal_min),
        )

        comprimento_interno = xlong2 - xlong1

        if comprimento_interno <= 0:
            continue

        # A linha precisa cobrir pelo menos 70% do comprimento do
        # contorno. O teste de fra��o acima j� limita o comprimento
        # absoluto; este segundo teste exige que ela tamb�m esteja
        # realmente sobre o contorno selecionado.
        fracao_sobreposicao = (
            sobreposicao / comprimento_longitudinal_pdf
        )

        if fracao_sobreposicao < 0.35:
            continue

        score_comprimento = min(
            fracao,
            1.0,
        ) * 60.0

        score_angulo = max(
            0.0,
            20.0 -
            diferenca_angular * 10.0,
        )

        score_borda = min(
            distancia_borda *
            escala /
            20.0,
            10.0,
        )

        score = (
            score_comprimento
            +
            score_angulo
            +
            score_borda
        )

        candidatos_brutos.append(
            {
                "segmento_indice":
                    segmento.indice,

                "coordenada_transversal_pdf":
                    coord_transversal,

                "comprimento_pdf":
                    comprimento_pdf,

                "comprimento_mm":
                    comprimento_mm,

                "fracao_comprimento":
                    fracao,

                "angulo_graus":
                    angulo,

                "diferenca_angular":
                    diferenca_angular,

                "score":
                    score,
            }
        )

    candidatos_brutos.sort(
        key=lambda item:
            item[
                "coordenada_transversal_pdf"
            ]
    )

    grupos = []

    tolerancia_agrupamento_pdf = (
        TOLERANCIA_AGRUPAMENTO_DOBRA_MM /
        escala
    )

    for candidato in candidatos_brutos:

        coord = candidato[
            "coordenada_transversal_pdf"
        ]

        grupo_encontrado = None

        for grupo in grupos:

            media = grupo[
                "media_coordenada_pdf"
            ]

            if abs(
                coord - media
            ) <= tolerancia_agrupamento_pdf:

                grupo["itens"].append(
                    candidato
                )

                total = len(
                    grupo["itens"]
                )

                grupo[
                    "media_coordenada_pdf"
                ] = (
                    sum(
                        item[
                            "coordenada_transversal_pdf"
                        ]
                        for item in grupo["itens"]
                    )
                    /
                    total
                )

                grupo_encontrado = grupo
                break

        if grupo_encontrado is None:

            grupos.append(
                {
                    "media_coordenada_pdf":
                        coord,

                    "itens":
                        [candidato],
                }
            )

    candidatos_agrupados = []

    for grupo in grupos:

        itens = grupo["itens"]

        comprimento_total_pdf = sum(
            item["comprimento_pdf"]
            for item in itens
        )

        comprimento_total_mm = (
            comprimento_total_pdf *
            escala
        )

        media_coord = grupo[
            "media_coordenada_pdf"
        ]

        melhor_score = max(
            item["score"]
            for item in itens
        )

        candidatos_agrupados.append(
            {
                "coordenada_transversal_pdf":
                    media_coord,

                "posicao_mm_desde_borda":
                    (
                        media_coord -
                        limite_transversal_min
                    ) *
                    escala,

                "comprimento_total_pdf":
                    comprimento_total_pdf,

                "comprimento_total_mm":
                    comprimento_total_mm,

                "quantidade_segmentos":
                    len(itens),

                "score":
                    melhor_score,

                "segmentos":
                    itens,
            }
        )

    candidatos_uteis = []

    for candidato in candidatos_agrupados:

        posicao = candidato[
            "posicao_mm_desde_borda"
        ]

        dimensao_transversal_mm = (
            limite_transversal_max -
            limite_transversal_min
        ) * escala

        distancia_borda_1 = posicao

        distancia_borda_2 = (
            dimensao_transversal_mm -
            posicao
        )

        if (
            distancia_borda_1 <
            MIN_DISTANCIA_BORDA_DOBRA_MM
        ):
            continue

        if (
            distancia_borda_2 <
            MIN_DISTANCIA_BORDA_DOBRA_MM
        ):
            continue

        candidatos_uteis.append(
            candidato
        )

    candidatos_uteis.sort(
        key=lambda item:
            item[
                "posicao_mm_desde_borda"
            ]
    )

    par_selecionado = None
    score_par = -999999.0

    raio = (
        raio_mm
        if raio_mm is not None
        else RAIO_PADRAO
    )

    espessura = (
        espessura_mm
        if espessura_mm is not None
        else ESPESSURA_PADRAO
    )

    setback_referencia = (
        (
            raio +
            espessura
        )
        *
        math.tan(
            math.radians(
                ANGULO_PADRAO / 2.0
            )
        )
    )

    dimensao_transversal_mm = (
        limite_transversal_max -
        limite_transversal_min
    ) * escala

    for i in range(
        len(candidatos_uteis)
    ):

        for j in range(
            i + 1,
            len(candidatos_uteis)
        ):

            c1 = candidatos_uteis[i]
            c2 = candidatos_uteis[j]

            p1 = c1[
                "posicao_mm_desde_borda"
            ]

            p2 = c2[
                "posicao_mm_desde_borda"
            ]

            trecho1 = p1
            trecho2 = p2 - p1
            trecho3 = (
                dimensao_transversal_mm -
                p2
            )

            if trecho1 <= 0:
                continue

            if trecho2 <= 0:
                continue

            if trecho3 <= 0:
                continue

            if trecho1 <= setback_referencia:
                continue

            if trecho2 <= 2.0 * setback_referencia:
                continue

            if trecho3 <= setback_referencia:
                continue

            score_atual = (
                c1["score"]
                +
                c2["score"]
            )

            distancia_entre = (
                p2 - p1
            )

            if (
                distancia_entre
                <
                3.0 * setback_referencia
            ):
                score_atual -= 20.0

            if score_atual > score_par:

                score_par = score_atual

                par_selecionado = (
                    c1,
                    c2,
                )

    encontradas = (
        par_selecionado is not None
    )

    linhas_selecionadas = []

    if encontradas:

        for numero, candidato in enumerate(
            par_selecionado,
            start=1,
        ):

            linhas_selecionadas.append(
                {
                    "numero":
                        numero,

                    "posicao_mm":
                        candidato[
                            "posicao_mm_desde_borda"
                        ],

                    "coordenada_pdf":
                        candidato[
                            "coordenada_transversal_pdf"
                        ],

                    "comprimento_mm":
                        candidato[
                            "comprimento_total_mm"
                        ],

                    "score":
                        candidato[
                            "score"
                        ],

                    "segmentos":
                        candidato[
                            "segmentos"
                        ],
                }
            )

    return {
        "encontradas":
            encontradas,

        "quantidade":
            len(linhas_selecionadas),

        "eixo_longitudinal":
            eixo,

        "eixo_desenvolvimento":
            (
                "TRANSVERSAL"
                if eixo == "X"
                else "LONGITUDINAL"
            ),

        "comprimento_longitudinal_mm":
            comprimento_longitudinal_mm,

        "dimensao_transversal_mm":
            dimensao_transversal_mm,

        "candidatos":
            candidatos_uteis,

        "linhas_selecionadas":
            linhas_selecionadas,

        "score_par":
            (
                score_par
                if encontradas
                else None
            ),

        "setback_referencia_mm":
            setback_referencia,

        "motivo":
            (
                "DUAS_LINHAS_INTERNAS_COMPATIVEIS"
                if encontradas
                else
                "LINHAS_DE_DOBRA_NAO_IDENTIFICADAS"
            ),
    }


# =============================================================================
# IDENTIFICAR DOBRAS
# =============================================================================

def identificar_dobras(
    textos: List[dict],
    contorno: Contorno,
    espessura_mm: Optional[float],
    todos_segmentos: List[Segmento],
) -> Tuple[
    List[Dobra],
    dict,
]:

    angulos_texto = extrair_angulos(
        textos
    )

    raios_texto = extrair_raios(
        textos
    )

    if len(angulos_texto) >= 2:

        angulos = angulos_texto[:2]
        origem_angulo = "TEXTO"

    elif len(angulos_texto) == 1:

        angulos = [
            angulos_texto[0],
            angulos_texto[0],
        ]

        origem_angulo = "TEXTO_E_REPETICAO"

    else:

        angulos = [
            ANGULO_PADRAO,
            ANGULO_PADRAO,
        ]

        origem_angulo = "CONFIGURACAO_PADRAO"

    if len(raios_texto) >= 2:

        raios = raios_texto[:2]
        origem_raio = "TEXTO"

    elif len(raios_texto) == 1:

        raios = [
            raios_texto[0],
            raios_texto[0],
        ]

        origem_raio = "TEXTO_E_REPETICAO"

    else:

        raios = [
            RAIO_PADRAO,
            RAIO_PADRAO,
        ]

        origem_raio = "CONFIGURACAO_PADRAO"

    raio_referencia = raios[0]

    deteccao_geometrica = (
        detectar_linhas_dobra_geometricas(
            todos_segmentos,
            contorno,
            espessura_mm,
            raio_referencia,
        )
    )

    dobras = []

    if (
        deteccao_geometrica["encontradas"]
        and
        deteccao_geometrica["quantidade"] >= 2
    ):

        linhas = (
            deteccao_geometrica[
                "linhas_selecionadas"
            ]
        )

        for i in range(2):

            angulo = (
                angulos[i]
                if i < len(angulos)
                else ANGULO_PADRAO
            )

            raio = (
                raios[i]
                if i < len(raios)
                else RAIO_PADRAO
            )

            linha = linhas[i]

            confianca = 0.97

            if (
                origem_angulo ==
                "CONFIGURACAO_PADRAO"
            ):
                confianca -= 0.03

            if (
                origem_raio ==
                "CONFIGURACAO_PADRAO"
            ):
                confianca -= 0.03

            dobras.append(
                Dobra(
                    numero=i + 1,
                    angulo=angulo,
                    raio=raio,
                    posicao_mm=(
                        linha[
                            "posicao_mm"
                        ]
                    ),
                    direcao=(
                        "PARALELA_EIXO_LONGITUDINAL"
                    ),
                    confianca=confianca,
                    origem=(
                        "GEOMETRIA_LINHA_INTERNA"
                    ),
                )
            )

    else:

        for i in range(
            QUANTIDADE_DOBRAS_PADRAO
        ):

            angulo = (
                angulos[i]
                if i < len(angulos)
                else ANGULO_PADRAO
            )

            raio = (
                raios[i]
                if i < len(raios)
                else RAIO_PADRAO
            )

            dobras.append(
                Dobra(
                    numero=i + 1,
                    angulo=angulo,
                    raio=raio,
                    posicao_mm=None,
                    direcao=None,
                    confianca=0.30,
                    origem=(
                        "DOBRA_IDENTIFICADA_SEM_POSICAO_GEOMETRICA"
                    ),
                )
            )

    return (
        dobras,
        deteccao_geometrica,
    )


# =============================================================================
# ORDENAR DOBRAS
# =============================================================================

def ordenar_dobras(
    dobras: List[Dobra],
) -> List[Dobra]:

    if not dobras:
        return []

    return sorted(
        dobras,
        key=lambda d: (
            d.posicao_mm
            if d.posicao_mm is not None
            else 999999
        ),
    )


# =============================================================================
# FATOR K
# =============================================================================

def fator_k_por_raio(
    raio: Optional[float],
    espessura: Optional[float],
) -> float:

    if espessura is None:
        return 0.33

    if raio is None:
        return 0.33

    relacao = (
        raio /
        espessura
    )

    if relacao <= 1.0:
        return 0.33

    if relacao <= 1.5:
        return 0.35

    if relacao <= 2.0:
        return 0.38

    if relacao <= 3.0:
        return 0.40

    return 0.42


# =============================================================================
# BEND ALLOWANCE
# =============================================================================

def bend_allowance(
    angulo: float,
    raio: float,
    espessura: float,
    k: float,
) -> float:

    theta = math.radians(
        abs(angulo)
    )

    return (
        theta *
        (
            raio +
            k * espessura
        )
    )


# =============================================================================
# BEND DEDUCTION
# =============================================================================

def bend_deduction(
    angulo: float,
    raio: float,
    espessura: float,
    k: float,
) -> float:

    theta_meio = math.radians(
        abs(angulo) / 2.0
    )

    ba = bend_allowance(
        angulo,
        raio,
        espessura,
        k,
    )

    return (
        2.0 *
        (
            raio +
            espessura
        )
        *
        math.tan(
            theta_meio
        )
        -
        ba
    )


# =============================================================================
# SETBACK
# =============================================================================

def bend_setback(
    angulo: float,
    raio: float,
    espessura: float,
) -> float:

    return (
        (
            raio +
            espessura
        )
        *
        math.tan(
            math.radians(
                abs(angulo) / 2.0
            )
        )
    )


# =============================================================================
# DESENVOLVIMENTO
# =============================================================================

def estimar_desenvolvimento(
    contorno: Contorno,
    dobras: List[Dobra],
    espessura_mm: Optional[float],
    deteccao_geometrica: dict,
) -> Tuple[
    float,
    dict,
]:

    """
    RECONSTRUÇÃO REAL:

        ABA 1
        BA 1
        ABA 2
        BA 2
        ABA 3

    A dimensão 672 mm NÃO entra como comprimento do desenvolvimento.

    Ela representa o comprimento longitudinal do blank.

    O desenvolvimento é reconstruído no eixo transversal.
    """

    if espessura_mm is None:
        espessura_mm = ESPESSURA_PADRAO

    if len(dobras) != 2:

        return (
            0.0,
            {
                "status":
                    "NAO_CALCULADO",

                "metodo":
                    "GEOMETRIA_INSUFICIENTE",

                "motivo":
                    "QUANTIDADE_DE_DOBRAS_DIFERENTE_DE_2",

                "base_mm":
                    0.0,

                "bend_allowance_total_mm":
                    0.0,

                "bend_deduction_total_mm":
                    0.0,

                "desenvolvimento_mm":
                    0.0,

                "detalhes":
                    [],

                "sequencia":
                    [],
            },
        )

    if any(
        dobra.posicao_mm is None
        for dobra in dobras
    ):

        return (
            0.0,
            {
                "status":
                    "NAO_CALCULADO",

                "metodo":
                    "GEOMETRIA_INSUFICIENTE",

                "motivo":
                    "POSICOES_REAIS_DAS_DOBRAS_NAO_IDENTIFICADAS",

                "base_mm":
                    0.0,

                "bend_allowance_total_mm":
                    0.0,

                "bend_deduction_total_mm":
                    0.0,

                "desenvolvimento_mm":
                    0.0,

                "detalhes":
                    [],
                "sequencia":
                    [],
            },
        )

    eixo = deteccao_geometrica.get(
        "eixo_longitudinal"
    )

    # -------------------------------------------------------------------------
    # A dimens?o transversal deve conter as posi??es reais das duas dobras.
    #
    # Isso ? mais confi?vel que usar diretamente X/Y do PDF, porque o
    # contorno pode ter sido normalizado/rotacionado durante a calibra??o.
    #
    # No componente 046:
    #
    #   largura = 672.0023
    #   altura  = 162.9995
    #   dobras  = 48.2704 / 114.7761
    #
    # Logo a transversal ? 162.9995 mm.
    # -------------------------------------------------------------------------

    posicoes_dobras_temp = sorted(
        [
            d.posicao_mm
            for d in dobras
            if d.posicao_mm is not None
        ]
    )

    if len(posicoes_dobras_temp) >= 2:

        maior_posicao_dobra = max(
            posicoes_dobras_temp
        )

        dimensoes_candidatas = [
            float(contorno.largura_mm),
            float(contorno.altura_mm),
        ]

        dimensoes_validas = [
            dimensao
            for dimensao in dimensoes_candidatas
            if dimensao >= maior_posicao_dobra
        ]

        if dimensoes_validas:

            dimensao_transversal_mm = min(
                dimensoes_validas
            )

        else:

            dimensao_transversal_mm = 0.0

    else:

        dimensao_transversal_mm = 0.0

    if dimensao_transversal_mm <= 0:

        return (
            0.0,
            {
                "status":
                    "NAO_CALCULADO",

                "metodo":
                    "GEOMETRIA_INSUFICIENTE",

                "motivo":
                    "DIMENSAO_TRANSVERSAL_NAO_IDENTIFICADA",

                "base_mm":
                    0.0,

                "bend_allowance_total_mm":
                    0.0,

                "bend_deduction_total_mm":
                    0.0,

                "desenvolvimento_mm":
                    0.0,

                "sequencia":
                    [],
            },
        )

    if eixo == "X":

        eixo_desenvolvimento = (
            "TRANSVERSAL_Y"
        )

    elif eixo == "Y":

        eixo_desenvolvimento = (
            "TRANSVERSAL_X"
        )

    else:

        return (
            0.0,
            {
                "status":
                    "NAO_CALCULADO",

                "metodo":
                    "GEOMETRIA_INSUFICIENTE",

                "motivo":
                    "EIXO_DAS_DOBRAS_NAO_IDENTIFICADO",

                "base_mm":
                    0.0,

                "bend_allowance_total_mm":
                    0.0,

                "bend_deduction_total_mm":
                    0.0,

                "desenvolvimento_mm":
                    0.0,

                "sequencia":
                    [],
            },
        )

    dobras_ordenadas = ordenar_dobras(
        dobras
    )

    p1 = dobras_ordenadas[0].posicao_mm
    p2 = dobras_ordenadas[1].posicao_mm

    if p1 is None or p2 is None:

        raise RuntimeError(
            "Posição de dobra ausente."
        )

    if p2 <= p1:

        return (
            0.0,
            {
                "status":
                    "NAO_CALCULADO",

                "metodo":
                    "GEOMETRIA_INCONSISTENTE",

                "motivo":
                    "ORDEM_DAS_DOBRAS_INVALIDA",

                "posicao_1_mm":
                    p1,

                "posicao_2_mm":
                    p2,

                "base_mm":
                    dimensao_transversal_mm,

                "sequencia":
                    [],
            },
        )

    if (
        p1 <= 0
        or
        p2 >= dimensao_transversal_mm
    ):

        return (
            0.0,
            {
                "status":
                    "NAO_CALCULADO",

                "metodo":
                    "GEOMETRIA_INCONSISTENTE",

                "motivo":
                    "POSICAO_DOBRA_FORA_DO_CONTORNO",

                "posicao_dobra_1_mm":
                    p1,

                "posicao_dobra_2_mm":
                    p2,

                "dimensao_transversal_mm":
                    dimensao_transversal_mm,

                "sequencia":
                    [],
            },
        )

    # -------------------------------------------------------------------------
    # PRIMEIRO: reconstrução geométrica
    # -------------------------------------------------------------------------

    perfil = reconstruir_perfil_transversal(
        contorno,
        dobras_ordenadas,
        espessura_mm,
    )

    # -------------------------------------------------------------------------
    # TRECHOS EXTERNOS
    # -------------------------------------------------------------------------

    trecho_externo_1 = p1

    trecho_entre_dobras = (
        p2 - p1
    )

    trecho_externo_3 = (
        dimensao_transversal_mm -
        p2
    )

    # -------------------------------------------------------------------------
    # BA / BD / SETBACK
    # -------------------------------------------------------------------------

    ba_total = 0.0
    bd_total = 0.0

    detalhes = []
    setbacks = []

    for dobra in dobras_ordenadas:

        raio = (
            dobra.raio
            if dobra.raio is not None
            else RAIO_PADRAO
        )

        k = fator_k_por_raio(
            raio,
            espessura_mm,
        )

        ba = bend_allowance(
            dobra.angulo,
            raio,
            espessura_mm,
            k,
        )

        bd = bend_deduction(
            dobra.angulo,
            raio,
            espessura_mm,
            k,
        )

        setback = bend_setback(
            dobra.angulo,
            raio,
            espessura_mm,
        )

        ba_total += ba
        bd_total += bd

        setbacks.append(
            setback
        )

        detalhes.append(
            {
                "dobra":
                    dobra.numero,

                "angulo_graus":
                    dobra.angulo,

                "raio_mm":
                    raio,

                "espessura_mm":
                    espessura_mm,

                "razao_raio_espessura":
                    (
                        raio /
                        espessura_mm
                    ),

                "fator_k":
                    k,

                "setback_mm":
                    setback,

                "bend_allowance_mm":
                    ba,

                "bend_deduction_mm":
                    bd,

                "posicao_mm":
                    dobra.posicao_mm,

                "origem_posicao":
                    dobra.origem,
            }
        )

    # -------------------------------------------------------------------------
    # ABAS
    # -------------------------------------------------------------------------

    aba_1 = (
        trecho_externo_1 -
        setbacks[0]
    )

    aba_2 = (
        trecho_entre_dobras -
        setbacks[0] -
        setbacks[1]
    )

    aba_3 = (
        trecho_externo_3 -
        setbacks[1]
    )

    # -------------------------------------------------------------------------
    # VALIDAÇÃO
    # -------------------------------------------------------------------------

    abas_validas = (
        aba_1 > 0
        and
        aba_2 > 0
        and
        aba_3 > 0
    )

    if not abas_validas:

        return (
            0.0,
            {
                "status":
                    "NAO_CALCULADO",

                "metodo":
                    "GEOMETRIA_INCONSISTENTE",

                "motivo":
                    "UMA_DAS_ABAS_E_NEGATIVA_OU_ZERO",

                "aba_1_mm":
                    aba_1,

                "aba_2_mm":
                    aba_2,

                "aba_3_mm":
                    aba_3,

                "perfil_transversal":
                    perfil,

                "detalhes":
                    detalhes,

                "sequencia":
                    [],
            },
        )

    # -------------------------------------------------------------------------
    # DESENVOLVIMENTO
    # -------------------------------------------------------------------------

    desenvolvimento = (
        aba_1
        +
        detalhes[0][
            "bend_allowance_mm"
        ]
        +
        aba_2
        +
        detalhes[1][
            "bend_allowance_mm"
        ]
        +
        aba_3
    )

    desenvolvimento_por_bd = (
        dimensao_transversal_mm -
        bd_total
    )

    diferenca_metodos = (
        desenvolvimento -
        desenvolvimento_por_bd
    )

    # -------------------------------------------------------------------------
    # SEQUÊNCIA EXPLÍCITA
    # -------------------------------------------------------------------------

    sequencia = [

        {
            "tipo":
                "ABA",

            "numero":
                1,

            "comprimento_mm":
                aba_1,

            "trecho_externo_mm":
                trecho_externo_1,

            "setback_mm":
                setbacks[0],

            "origem":
                "GEOMETRIA_ABA_1",
        },

        {
            "tipo":
                "BA",

            "dobra":
                1,

            "comprimento_mm":
                detalhes[0][
                    "bend_allowance_mm"
                ],

            "angulo_graus":
                detalhes[0][
                    "angulo_graus"
                ],

            "raio_mm":
                detalhes[0][
                    "raio_mm"
                ],

            "fator_k":
                detalhes[0][
                    "fator_k"
                ],
        },

        {
            "tipo":
                "ABA",

            "numero":
                2,

            "comprimento_mm":
                aba_2,

            "trecho_externo_mm":
                trecho_entre_dobras,

            "setback_1_mm":
                setbacks[0],

            "setback_2_mm":
                setbacks[1],

            "origem":
                "GEOMETRIA_ABA_2",
        },

        {
            "tipo":
                "BA",

            "dobra":
                2,

            "comprimento_mm":
                detalhes[1][
                    "bend_allowance_mm"
                ],

            "angulo_graus":
                detalhes[1][
                    "angulo_graus"
                ],

            "raio_mm":
                detalhes[1][
                    "raio_mm"
                ],

            "fator_k":
                detalhes[1][
                    "fator_k"
                ],
        },

        {
            "tipo":
                "ABA",

            "numero":
                3,

            "comprimento_mm":
                aba_3,

            "trecho_externo_mm":
                trecho_externo_3,

            "setback_mm":
                setbacks[1],

            "origem":
                "GEOMETRIA_ABA_3",
        },
    ]

    soma_sequencia = sum(
        item[
            "comprimento_mm"
        ]
        for item in sequencia
    )

    erro_fechamento = (
        desenvolvimento -
        soma_sequencia
    )

    # -------------------------------------------------------------------------
    # CONFERÊNCIA DO PERFIL GEOMÉTRICO
    # -------------------------------------------------------------------------

    perfil_confirmado = (
        perfil.get(
            "status"
        )
        in (
            "CONFIRMADO",
            "PARCIAL",
        )
    )

    deteccao_confirmada = (
        deteccao_geometrica.get("encontradas", False)
        and deteccao_geometrica.get("quantidade", 0) == 2
    )

    perfil_nao_inconsistente = (
        perfil.get("status")
        not in ("GEOMETRIA_INCONSISTENTE",)
    )

    valido = (
        abs(
            erro_fechamento
        )
        <= TOLERANCIA_RECONSTRUCAO_MM
        and
        desenvolvimento > 0
        and
        deteccao_confirmada
        and
        perfil_nao_inconsistente
    )

    return (
        desenvolvimento,
        {
            "status":
                (
                    "CALCULADO"
                    if valido
                    else
                    "NAO_CALCULADO"
                ),

            "metodo":
                "GEOMETRIA_ABA_BA_ABA_BA_ABA",

            "eixo_desenvolvimento":
                eixo_desenvolvimento,

            "eixo_longitudinal":
                eixo,

            "base_mm":
                dimensao_transversal_mm,

            "dimensao_transversal_mm":
                dimensao_transversal_mm,

            "posicao_dobra_1_mm":
                p1,

            "posicao_dobra_2_mm":
                p2,

            "trecho_externo_1_mm":
                trecho_externo_1,

            "trecho_entre_dobras_mm":
                trecho_entre_dobras,

            "trecho_externo_3_mm":
                trecho_externo_3,

            "setback_1_mm":
                setbacks[0],

            "setback_2_mm":
                setbacks[1],

            "aba_1_mm":
                aba_1,

            "aba_2_mm":
                aba_2,

            "aba_3_mm":
                aba_3,

            "bend_allowance_total_mm":
                ba_total,

            "bend_deduction_total_mm":
                bd_total,

            "desenvolvimento_mm":
                desenvolvimento,

            "desenvolvimento_por_bd_mm":
                desenvolvimento_por_bd,

            "diferenca_entre_metodos_mm":
                diferenca_metodos,

            "soma_sequencia_mm":
                soma_sequencia,

            "erro_fechamento_mm":
                erro_fechamento,

            "detalhes":
                detalhes,

            "sequencia":
                sequencia,

            "perfil_transversal":
                perfil,

            "deteccao_geometrica":
                deteccao_geometrica,

            "perfil_geometrico_confirmado":
                perfil_confirmado,

            "criterio":
                (
                    "SEGMENTOS_LONGITUDINAIS_REAIS"
                    "_DOBRAS_GEOMETRICAS"
                    "_ABA_BA_ABA_BA_ABA"
                ),

            "deteccao_dobras_confirmada":
                deteccao_confirmada,

            "nao_utiliza_bounding_box_para_desenvolvimento":
                True,
        },
    )

# BLANK
# =============================================================================

def calcular_blank(
    contorno: Contorno,
    desenvolvimento_mm: float,
    espessura_mm: Optional[float],
    desenvolvimento_detalhes: dict,
) -> Blank:

    status = (
        desenvolvimento_detalhes.get(
            "status",
            "NAO_CALCULADO",
        )
    )

    if (
        status != "CALCULADO"
        or
        desenvolvimento_mm <= 0
    ):

        return Blank(
            comprimento_desenvolvido_mm=0.0,
            largura_blank_mm=0.0,
            comprimento_blank_mm=0.0,
            espessura_mm=espessura_mm,
            metodo=(
                "NAO_CALCULADO_GEOMETRIA_INSUFICIENTE"
            ),
            valido=False,
        )

    eixo_longitudinal = (
        desenvolvimento_detalhes.get(
            "eixo_longitudinal"
        )
    )

    if eixo_longitudinal == "X":

        comprimento_longitudinal = (
            contorno.largura_mm
        )

    elif eixo_longitudinal == "Y":

        comprimento_longitudinal = (
            contorno.altura_mm
        )

    else:
        return Blank(
            comprimento_desenvolvido_mm=0.0,
            largura_blank_mm=0.0,
            comprimento_blank_mm=0.0,
            espessura_mm=espessura_mm,
            metodo="NAO_CALCULADO_EIXO_LONGITUDINAL_INVALIDO",
            valido=False,
        )

    return Blank(
        comprimento_desenvolvido_mm=
            desenvolvimento_mm,

        largura_blank_mm=
            desenvolvimento_mm,

        comprimento_blank_mm=
            comprimento_longitudinal,

        espessura_mm=
            espessura_mm,

        metodo=
            "DESENVOLVIMENTO_GEOMETRICO_ABAS_BA",

        valido=True,
    )


# =============================================================================
# DETECTAR CONTORNOS
# =============================================================================

def detectar_contornos(
    segmentos: List[Segmento],
) -> List[Contorno]:

    grupos = agrupar_segmentos_conectados(
        segmentos
    )

    contornos = []

    for i, grupo in enumerate(
        grupos
    ):

        if len(grupo) < 3:
            continue

        contorno = criar_contorno(
            grupo,
            f"{i + 1:03d}",
        )

        if contorno is None:
            continue

        contornos.append(
            contorno
        )

    contornos.sort(
        key=lambda c:
            c.area_pdf,
        reverse=True,
    )

    for i, contorno in enumerate(
        contornos,
        start=1,
    ):

        contorno.componente = (
            f"{i:03d}"
        )

    return contornos

) -> Optional[Contorno]:

    for contorno in contornos:

        if (
            contorno.largura_pdf <= 0
            or
            contorno.altura_pdf <= 0
        ):
            continue

        pontuar_contorno(
            contorno,
            DIMENSAO_ALVO_LARGURA,
            DIMENSAO_ALVO_ALTURA,
        )

        melhores.append(
            contorno
        )

    if not melhores:
        return None

        DIMENSAO_ALVO_LARGURA /
        DIMENSAO_ALVO_ALTURA
    )

    def erro_proporcao(c):

        proporcao = (
            max(
                c.largura_pdf,
                c.altura_pdf,
            )
            /
            max(
                min(
                    c.largura_pdf,
                    c.altura_pdf,
                ),
                0.0001,
            )
        )

        return abs(
            proporcao -
            alvo_proporcao
        )

    melhores.sort(
        key=lambda c: (
            -c.score,
            erro_proporcao(c),
        )
    )

    candidato = melhores[0]

    proporcao = (
        max(
            candidato.largura_pdf,
