            candidato.altura_pdf,
        )
        /
        max(
            min(
                candidato.largura_pdf,
                candidato.altura_pdf,
            ),
            0.0001,
        )
    )

    if proporcao < 2.5:

        alongados = [
            c
            for c in melhores
            if (
                max(
                    c.largura_pdf,
                    c.altura_pdf,
                )
                /
                max(
                    min(
                        c.largura_pdf,
                        c.altura_pdf,
                    ),
                    0.0001,
                )
                >= 2.5
            )
        ]

        if alongados:

            candidato = alongados[0]

    candidato.componente = "046"

    return candidato


# =============================================================================
# CLASSIFICAR SEGMENTO
# =============================================================================

def classificar_segmento(
    segmento: Segmento,
) -> str:

    comprimento = (
        segmento.comprimento
    )

    if comprimento < 0.5:
        return "AUXILIAR"

    angulo = (
        abs(
            segmento.angulo
        )
        %
        180
    )

    if (
        abs(angulo) < 1
        or
        abs(angulo - 180) < 1
        or
        abs(angulo - 90) < 1
    ):

        return "RETO"

    return "RETO_INCLINADO"


# =============================================================================
# SEGMENTOS EM MM
# =============================================================================

def segmentos_em_mm(
    contorno: Contorno,
) -> List[dict]:

    resultado = []

    for i, s in enumerate(
        contorno.segmentos,
        start=1,
    ):

        resultado.append(
            {
                "segmento":
                    i,

                "x1_mm":
                    round(
                        s.x1 *
                        contorno.escala,
                        4,
                    ),

                "y1_mm":
                    round(
                        s.y1 *
                        contorno.escala,
                        4,
                    ),

                "x2_mm":
                    round(
                        s.x2 *
                        contorno.escala,
                        4,
                    ),

                "y2_mm":
                    round(
                        s.y2 *
                        contorno.escala,
                        4,
                    ),

                "comprimento_mm":
                    round(
                        s.comprimento *
                        contorno.escala,
                        4,
                    ),

                "angulo_graus":
                    round(
                        s.angulo,
                        4,
                    ),

                "tipo":
                    classificar_segmento(
                        s
                    ),
            }
        )

    return resultado

# =============================================================================
# PROCESSAMENTO PRINCIPAL
# =============================================================================

def processar(
    caminho_pdf: str,
):

    caminho = Path(
        caminho_pdf
    )

    if not caminho.exists():

        raise FileNotFoundError(
            f"Arquivo não encontrado: "
            f"{caminho}"
        )

    print("=" * 80)
    print(
        "AIZI ENGINEERING AI"
    )
    print(
        "DIAGNÓSTICO DE DESENHO"
    )
    print("=" * 80)

    print(
        f"Arquivo: {caminho}"
    )

    documento = pymupdf.open(
        caminho
    )

    todos_segmentos = []
    todos_textos = []

    for numero_pagina, pagina in enumerate(
        documento,
        start=1,
    ):

        print(
            f"Processando página "
            f"{numero_pagina}..."
        )

        segmentos = extrair_linhas(
            pagina
        )

        textos = extrair_textos(
            pagina
        )

        todos_segmentos.extend(
            segmentos
        )

        todos_textos.extend(
            textos
        )

    documento.close()

    print(
        f"Linhas vetoriais encontradas: "
        f"{len(todos_segmentos)}"
    )

    print(
        f"Textos encontrados: "
        f"{len(todos_textos)}"
    )

    # -------------------------------------------------------------------------
    # CONTORNOS
    # -------------------------------------------------------------------------

    contornos = detectar_contornos(
        todos_segmentos
    )

    print(
        f"Contornos encontrados: "
        f"{len(contornos)}"
    )

    if not contornos:

        raise RuntimeError(
            "Nenhum contorno fechado encontrado."
        )

    # -------------------------------------------------------------------------
    # COMPONENTE 046
    # -------------------------------------------------------------------------

    principal = selecionar_046(
        contornos
    )

    if principal is None:

        raise RuntimeError(
            "Não foi possível identificar "
            "o contorno principal."
        )

    principal.componente = "046"

    # -------------------------------------------------------------------------
    # ESPESSURA
    # -------------------------------------------------------------------------

    espessura = identificar_espessura(
        todos_textos
    )

    if espessura is None:

        espessura = ESPESSURA_PADRAO

    # -------------------------------------------------------------------------
    # DOBRAS
    # -------------------------------------------------------------------------

    dobras, deteccao_geometrica = (
        identificar_dobras(
            todos_textos,
            principal,
            espessura,
            todos_segmentos,
        )
    )

    dobras = ordenar_dobras(
        dobras
    )

    # -------------------------------------------------------------------------
    # DESENVOLVIMENTO
    # -------------------------------------------------------------------------

    desenvolvimento, detalhes = (
        estimar_desenvolvimento(
            principal,
            dobras,
            espessura,
            deteccao_geometrica,
        )
    )

    # -------------------------------------------------------------------------
    # BLANK
    # -------------------------------------------------------------------------

    blank = calcular_blank(
        principal,
        desenvolvimento,
        espessura,
        detalhes,
    )

    # -------------------------------------------------------------------------
    # PASTA DE SAÍDA
    # -------------------------------------------------------------------------

    pasta_saida = (
        caminho.parent /
        caminho.stem
    )

    pasta_saida.mkdir(
        parents=True,
        exist_ok=True,
    )

    caminho_txt = (
        pasta_saida /
        "_diagnostico.txt"
    )

    caminho_json = (
        pasta_saida /
        "_calculadora_corte.json"
    )

    # -------------------------------------------------------------------------
    # TXT
    # -------------------------------------------------------------------------

    gerar_relatorio(
        caminho,
        caminho_txt,
        principal,
        dobras,
        blank,
        espessura,
        detalhes,
        todos_textos,
    )

    # -------------------------------------------------------------------------
    # JSON
    # -------------------------------------------------------------------------

    gerar_json_calculadora_corte(
        caminho_json,
        principal,
        dobras,
        blank,
        espessura,
        detalhes,
    )

    # -------------------------------------------------------------------------
    # RESULTADO
    # -------------------------------------------------------------------------

    imprimir_resultado(
        principal,
        dobras,
        blank,
        caminho_txt,
        caminho_json,
        detalhes,
    )

    return {

        "componente":
            "046",

        "largura_mm":
            principal.largura_mm,

        "altura_mm":
            principal.altura_mm,

        "area_mm2":
            principal.area_mm2,

        "perimetro_mm":
            principal.perimetro_mm,

        "segmentos":
            len(
                principal.segmentos
            ),

        "dobras":
            [
                asdict(d)
                for d in dobras
            ],

        "deteccao_geometrica":
            deteccao_geometrica,

        "desenvolvimento_mm":
            blank.comprimento_desenvolvido_mm,

        "desenvolvimento_detalhes":
            detalhes,

        "blank": {

            "largura_mm":
                blank.largura_blank_mm,

            "comprimento_mm":
                blank.comprimento_blank_mm,

            "espessura_mm":
                blank.espessura_mm,

            "valido":
                blank.valido,
        },

        "diagnostico":
            str(caminho_txt),

        "calculadora_corte":
            str(caminho_json),
    }

# LOCALIZAR PDF
# =============================================================================

def selecionar_pdf() -> Optional[str]:

    try:

        import tkinter as tk
        from tkinter import filedialog

        root = tk.Tk()

        root.withdraw()

        caminho = (
            filedialog.askopenfilename(
                title=
                    "Selecione o desenho técnico",

                filetypes=[
                    (
                        "Arquivos PDF",
                        "*.pdf",
                    ),
                    (
                        "Todos os arquivos",
                        "*.*",
                    ),
                ],
            )
        )

        root.destroy()

        if caminho:
            return caminho

    except Exception as erro:

        print(
            f"Não foi possível abrir seletor: "
            f"{erro}"
        )

    return None
